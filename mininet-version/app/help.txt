# 1. Analizza i flussi e genera la topologia JSON
python app/analyze_flows.py features/target.csv topology_with_flows.json

# 2. Replay della topologia in Mininet (dentro il container)
docker compose run --rm mininet python /app/replay_topology.py /shared/topology_with_flows.json 60

# 3. Durante il replay, cattura il traffico
tcpdump -i any -w /pcap/replay_capture.pcap

# 4. Analizza il traffico catturato con CICFlowMeter
docker compose run --rm cicflowmeter ./cicflowmeter.sh /pcap/replay_capture.pcap /features/replay_features.csv
